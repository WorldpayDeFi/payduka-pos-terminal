<script>
// ─── CONFIG ───────────────────────────────────────────────────────────────────
const CONFIG = {
  ICO_CONTRACT:   "0xD8358ba36723b860FFC8E9830060a8Cf348B84d3",
  PDUKA_TOKEN:    "0x5025053F6Dec8C8Cfa85AE24C0CB9e4935C98E54",
  USDC_CONTRACT:  "0x2791Bca1f2de4661ED88A30C99A7a9449Aa84174", // USDC on Polygon
  POLYGON_CHAIN_ID: 137,
  SEED_PRICE_USD: 0.005,
};

const ICO_ABI = [
  "function buyWithUSDC(uint256 usdcAmount) external",
  "function buyWithPOL() external payable",
  "function totalRaised() external view returns (uint256)",
  "function tokensSold() external view returns (uint256)",
  "function tokensAvailable() external view returns (uint256)",
];

const ERC20_ABI = [
  "function approve(address spender, uint256 amount) external returns (bool)",
  "function allowance(address owner, address spender) external view returns (uint256)",
  "function balanceOf(address account) external view returns (uint256)",
];

// ─── STATE ────────────────────────────────────────────────────────────────────
let provider, signer, walletAddress;
let activeTab = 'usdc';
let polPriceUSD = 0.6; // fallback; will fetch live

// ─── INIT ─────────────────────────────────────────────────────────────────────
window.addEventListener('load', async () => {
  await fetchPolPrice();
  await loadICOStats();
  // animate progress bar after load
  setTimeout(() => {
    const bar = document.getElementById('progress-bar');
    const pct = parseFloat(bar.dataset.pct || 0);
    bar.style.width = pct + '%';
  }, 300);

  // auto-reconnect if already approved
  if (window.ethereum?.selectedAddress) {
    await connectWallet(true);
  }
});

async function fetchPolPrice() {
  try {
    const r = await fetch('https://api.coingecko.com/api/v3/simple/price?ids=matic-network&vs_currencies=usd');
    const d = await r.json();
    polPriceUSD = d['matic-network']?.usd || 0.6;
  } catch { /* use fallback */ }
}

async function loadICOStats() {
  try {
    const rpc = new ethers.JsonRpcProvider('https://polygon-rpc.com');
    const ico = new ethers.Contract(CONFIG.ICO_CONTRACT, ICO_ABI, rpc);
    const [raised, sold, avail] = await Promise.all([
      ico.totalRaised().catch(() => 0n),
      ico.tokensSold().catch(() => 0n),
      ico.tokensAvailable().catch(() => 1_000_000_000n * 10n**18n),
    ]);

    const raisedUSD   = Number(raised) / 1e6; // USDC 6 decimals
    const soldTokens  = Number(sold) / 1e18;
    const totalTokens = 1_000_000_000;
    const pct = Math.min((soldTokens / totalTokens) * 100, 100);

    document.getElementById('stat-raised').textContent    = '$' + fmtNum(raisedUSD);
    document.getElementById('stat-available').textContent = fmtCompact(Number(avail) / 1e18);
    document.getElementById('pct-text').textContent       = pct.toFixed(2) + '%';

    const bar = document.getElementById('progress-bar');
    bar.dataset.pct = pct;
  } catch (e) {
    console.warn('Stats fetch failed:', e);
  }
}

// ─── WALLET ───────────────────────────────────────────────────────────────────
async function connectWallet(silent = false) {
  if (!window.ethereum) {
    if (!silent) showToast('No wallet found. Install MetaMask or use a Web3 browser.', 'error');
    return;
  }
  try {
    provider = new ethers.BrowserProvider(window.ethereum);
    await provider.send('eth_requestAccounts', []);

    // Check network
    const network = await provider.getNetwork();
    if (Number(network.chainId) !== CONFIG.POLYGON_CHAIN_ID) {
      await switchToPolygon();
    }

    signer = await provider.getSigner();
    walletAddress = await signer.getAddress();

    document.getElementById('section-connect').style.display = 'none';
    document.getElementById('section-buy').style.display = 'block';
    document.getElementById('wallet-addr').textContent =
      walletAddress.slice(0, 6) + '...' + walletAddress.slice(-4) + ' ✓';

    if (!silent) showToast('Wallet connected!', 'success');
  } catch (e) {
    if (!silent) showToast('Connection failed: ' + (e.message || 'Unknown error'), 'error');
  }
}

async function switchToPolygon() {
  try {
    await window.ethereum.request({
      method: 'wallet_switchEthereumChain',
      params: [{ chainId: '0x89' }],
    });
  } catch (e) {
    if (e.code === 4902) {
      await window.ethereum.request({
        method: 'wallet_addEthereumChain',
        params: [{
          chainId: '0x89',
          chainName: 'Polygon Mainnet',
          rpcUrls: ['https://polygon-rpc.com'],
          nativeCurrency: { name: 'POL', symbol: 'POL', decimals: 18 },
          blockExplorerUrls: ['https://polygonscan.com'],
        }],
      });
    }
  }
}

// ─── TABS ─────────────────────────────────────────────────────────────────────
function switchTab(tab) {
  activeTab = tab;
  document.getElementById('tab-usdc').classList.toggle('active', tab === 'usdc');
  document.getElementById('tab-pol').classList.toggle('active', tab === 'pol');
  document.getElementById('pay-tag').textContent = tab.toUpperCase();
  document.getElementById('pay-amount').value = '';
  document.getElementById('recv-amount').textContent = '0 PDUKA';
  document.getElementById('recv-usd').textContent = '≈ $0.00';
}

// ─── CALC ─────────────────────────────────────────────────────────────────────
function calcTokens() {
  const amount = parseFloat(document.getElementById('pay-amount').value) || 0;
  let usdValue = activeTab === 'usdc' ? amount : amount * polPriceUSD;
  const tokens = usdValue / CONFIG.SEED_PRICE_USD;
  document.getElementById('recv-amount').textContent = fmtNum(tokens) + ' PDUKA';
  document.getElementById('recv-usd').textContent    = '≈ $' + usdValue.toFixed(2);
}

// ─── BUY ──────────────────────────────────────────────────────────────────────
async function buyTokens() {
  const amount = parseFloat(document.getElementById('pay-amount').value);
  if (!amount || amount <= 0) { showToast('Enter a valid amount.', 'error'); return; }

  const btn = document.getElementById('btn-buy');
  btn.disabled = true;
  btn.innerHTML = '<span class="spinner"></span>';

  try {
    const ico = new ethers.Contract(CONFIG.ICO_CONTRACT, ICO_ABI, signer);

    if (activeTab === 'usdc') {
      const usdc      = new ethers.Contract(CONFIG.USDC_CONTRACT, ERC20_ABI, signer);
      const amountWei = ethers.parseUnits(amount.toString(), 6);

      showToast('Approving USDC...', 'info');
      const allowance = await usdc.allowance(walletAddress, CONFIG.ICO_CONTRACT);
      if (allowance < amountWei) {
        const approveTx = await usdc.approve(CONFIG.ICO_CONTRACT, amountWei);
        await approveTx.wait();
      }

      showToast('Sending transaction...', 'info');
      const tx = await ico.buyWithUSDC(amountWei);
      await tx.wait();

    } else {
      const amountWei = ethers.parseEther(amount.toString());
      showToast('Sending transaction...', 'info');
      const tx = await ico.buyWithPOL({ value: amountWei });
      await tx.wait();
    }

    showToast('🎉 Purchase successful! PDuka tokens are in your wallet.', 'success');
    document.getElementById('pay-amount').value = '';
    calcTokens();
    await loadICOStats();

  } catch (e) {
    const msg = e?.reason || e?.message || 'Transaction failed';
    showToast('Error: ' + msg.slice(0, 80), 'error');
  } finally {
    btn.disabled = false;
    btn.innerHTML = 'Buy PDuka Tokens';
  }
}

// ─── UTILS ────────────────────────────────────────────────────────────────────
function fmtNum(n) {
  if (n >= 1_000_000) return (n / 1_000_000).toFixed(2) + 'M';
  if (n >= 1_000)     return (n / 1_000).toFixed(1) + 'K';
  return n.toFixed(2);
}

function fmtCompact(n) {
  if (n >= 1_000_000_000) return (n / 1_000_000_000).toFixed(2) + 'B';
  if (n >= 1_000_000)     return (n / 1_000_000).toFixed(1) + 'M';
  return n.toFixed(0);
}

function copyAddr(addr) {
  navigator.clipboard.writeText(addr).then(() => showToast('Address copied!', 'info'));
}

let toastTimer;
function showToast(msg, type = 'info') {
  const t = document.getElementById('toast');
  t.textContent = msg;
  t.className = 'toast ' + type + ' show';
  clearTimeout(toastTimer);
  toastTimer = setTimeout(() => t.classList.remove('show'), 4000);
}
